void main(){
        // Screen texel coordinates
        ivec2 screenTexelCoord = ivec2(gl_FragCoord.xy);

        bool realSky = false;

        float depth = texelFetch(depthtex0, screenTexelCoord, 0).x;

        // Distant Horizons apparently uses a different depth texture
        #ifdef DISTANT_HORIZONS
            realSky = depth == 1;
            if(realSky) depth = texelFetch(dhDepthTex0, screenTexelCoord, 0).x;
        #endif

        // Get screen pos
        vec3 screenPos = vec3(texCoord, depth);
        
        // Distant Horizons apparently uses a different projection matrix
        #ifdef DISTANT_HORIZONS
            vec3 viewPos = getViewPos(realSky ? dhProjectionInverse : gbufferProjectionInverse, screenPos);
        #else
            vec3 viewPos = getViewPos(gbufferProjectionInverse, screenPos);
        #endif

        // Get eye player pos
        vec3 eyePlayerPos = mat3(gbufferModelViewInverse) * viewPos;
        // Get feet player pos
        vec3 feetPlayerPos = eyePlayerPos + gbufferModelViewInverse[3].xyz;

        // Get scene color
        sceneColOut = texelFetch(colortex4, screenTexelCoord, 0).rgb;

        // ================================================================
        // ECLIPSE PHYSICAL WATER ABSORPTION + CAUSTICS
        // ================================================================
        {
            vec3 matRaw = texelFetch(colortex3, screenTexelCoord, 0).xyz;
            float smoothnessSample = matRaw.y;
            float matFlag = matRaw.z;
            bool isWaterMaterial = (matFlag > 0.45 && matFlag < 0.55);
            bool eyeInWater = (isEyeInWater == 1);
            bool isWaterPixel = isWaterMaterial && smoothnessSample > 0.75;

            vec3 WsunVec = normalize(vec3(shadowModelView[0].z, shadowModelView[1].z, shadowModelView[2].z));
            vec3 totEpsilon = vec3(Water_Absorb_R, Water_Absorb_G, Water_Absorb_B)
                            + Dirt_Amount * vec3(Dirt_Absorb_R, Dirt_Absorb_G, Dirt_Absorb_B);

            if (isWaterPixel) {
                // Water surface position — use correct projection for DH vs close
                #ifdef DISTANT_HORIZONS
                    bool isDHWater = (texelFetch(depthtex0, screenTexelCoord, 0).x >= 1.0);
                    vec3 waterSurfaceViewPos = isDHWater
                        ? getViewPos(dhProjectionInverse, vec3(texCoord, depth))
                        : getViewPos(gbufferProjectionInverse, vec3(texCoord, depth));
                #else
                    const bool isDHWater = false;
                    vec3 waterSurfaceViewPos = getViewPos(gbufferProjectionInverse, vec3(texCoord, depth));
                #endif
                vec3 waterSurfacePos = mat3(gbufferModelViewInverse) * waterSurfaceViewPos;
                vec3 worldPos = waterSurfacePos + cameraPosition;

                // Terrain behind water — use DH depth for DH pixels, regular for close
                float terrainDepth = texelFetch(depthtex1, screenTexelCoord, 0).x;
                vec3 terrainViewPos;
                #ifdef DISTANT_HORIZONS
                    if (isDHWater || terrainDepth >= 1.0) {
                        float dhTerrainDepth = texelFetch(dhDepthTex1, screenTexelCoord, 0).x;
                        terrainViewPos = getViewPos(dhProjectionInverse, vec3(texCoord, dhTerrainDepth));
                    } else {
                        terrainViewPos = getViewPos(gbufferProjectionInverse, vec3(texCoord, terrainDepth));
                    }
                #else
                    terrainViewPos = getViewPos(gbufferProjectionInverse, vec3(texCoord, terrainDepth));
                #endif
                vec3 terrainPlayerPos = mat3(gbufferModelViewInverse) * terrainViewPos;
                float Vdiff = max(distance(waterSurfacePos, terrainPlayerPos), 0.1);

                // Beer-Lambert absorption — stays bright until 3 blocks deep, then darkens
                float depthFactor = 1.0 + max(Vdiff - 3.0, 0.0) * 0.4;
                vec3 Absorbtion = exp(-totEpsilon * Vdiff * depthFactor);

                // Deep water body colour — scatter colour = what ISN'T absorbed
                // exp(-eps*depth) = survival fraction; (1 - survival) = what scatters back = water colour
                // When R_absorb=0: red survival=1, scatter=0 → no red in water (correct)
                // When R_absorb=0.39: red survival=0.08 at depth 6, scatter=0.92 → lots of red absorbed → blue/teal water
                #ifndef FORCE_DISABLE_DAY_CYCLE
                    float dayAmt = dayCycleAdjust;
                #else
                    float dayAmt = 1.0;
                #endif
                float bodyDepth = max(Vdiff * 2.0, 4.0);
                // Water body colour = what light SURVIVES the water column
                // exp(-eps*depth): R absorbed fast → low survival → dark red channel
                // B absorbed slow → high survival → bright blue channel → teal/blue water
                vec3 waterBodyCol = exp(-totEpsilon * bodyDepth) * dayAmt * 6.0;

                // Core Beer-Lambert composite:
                // shallow → terrain tinted by absorption
                // deep    → pure water body colour
                // colortex6 = pre-water opaque snapshot (written by deferred1 before water renders)
                // Wave-driven refraction using actual water normals
                vec3 waterNormal = texelFetch(colortex1, screenTexelCoord, 0).xyz;
                vec2 refractOffset = vec2(0.0);
                if (length(waterNormal) > 0.0001 && isEyeInWater == 0) {
                    vec3 nView = normalize(mat3(gbufferModelView) * waterNormal);
                    float dist = length(viewPos);
                    // Use wave normals to drive refraction
                    refractOffset = nView.xy * (0.18 / (1.0 + dist * 0.35));
                }
                vec2 refractUV = clamp(texCoord + refractOffset, vec2(0.001), vec2(0.999));
                if (textureLod(depthtex0, refractUV, 0).r >= 1.0) refractUV = texCoord;

                // Stronger chromatic effect with wave-driven refraction
                vec2 chromaOff = refractOffset * 1.8;
                vec3 terrainBase = vec3(
                    textureLod(colortex6, clamp(refractUV + chromaOff * 0.4,  vec2(0.001), vec2(0.999)), 0).r,
                    textureLod(colortex6, refractUV, 0).g,
                    textureLod(colortex6, clamp(refractUV - chromaOff * 0.3, vec2(0.001), vec2(0.999)), 0).b
                );

                // Preserve sun/moon highlights so bloom still has a source.
                vec3 straightBase = textureLod(colortex6, texCoord, 0).rgb;
                terrainBase = max(terrainBase, straightBase * 0.0);
                float straightLuma = dot(straightBase, vec3(0.299, 0.587, 0.114));
                if (straightLuma > 2.0) terrainBase = max(terrainBase, straightBase);

                sceneColOut = terrainBase * Absorbtion + waterBodyCol * (vec3(1.0) - Absorbtion);
            }

            // Underwater absorption — very dark, short visibility
            if (eyeInWater) {
                float distFromSurface = max(-(eyePlayerPos.y + cameraPosition.y - waterEnteredAltitude), 0.0);

                float uwDepthFactor = 1.0 + max(distFromSurface - 1.0, 0.0) * 2.5;
                vec3 uwAbsorption = exp(-totEpsilon * distFromSurface * uwDepthFactor);
                #ifndef FORCE_DISABLE_DAY_CYCLE
                    float uwDayAmt = dayCycleAdjust;
                #else
                    float uwDayAmt = 1.0;
                #endif
                float uwBodyDepth = max(distFromSurface * 2.0, 4.0);
                vec3 uwWaterBodyCol = exp(-totEpsilon * uwBodyDepth) * uwDayAmt * 0.8;

                // Short view distance fog — exponential falloff, ~4 block visibility
                float viewDepth = length(viewPos);
                float visibilityFog = 1.0 - exp(-viewDepth * 0.35);

                sceneColOut = sceneColOut * uwAbsorption * 0.08 + uwWaterBodyCol * (vec3(1.0) - uwAbsorption);
                sceneColOut = mix(sceneColOut, vec3(0.0), visibilityFog);

                if (WsunVec.y > 0.0) {
                    #ifndef FORCE_DISABLE_DAY_CYCLE
                        float cs = clamp(dayCycleAdjust * 2.0, 0.0, 1.0);
                    #else
                        float cs = 1.0;
                    #endif
                    // Use per-pixel world position (terrain surface under water)
                    // so caustic appears as a pattern on the floor/walls, not a uniform flash
                    vec3 pixelWorldPos = feetPlayerPos + cameraPosition;
                    float caustic = waterCaustics(pixelWorldPos, WsunVec, distFromSurface);
                    
                    // Tie caustics to waves - sample with same tiling scale as caustics
                    float animateTime = frameTimeCounter * CURRENT_SPEED * 0.0625;
                    // Match caustic tiling: scale = 20.0 * waterTileSizeInv, but multiply for finer detail
                    const float causticScale = (20.0 * 3.0) * waterTileSizeInv;
                    vec2 waveCoord = vec2(pixelWorldPos.x * causticScale, pixelWorldPos.z * causticScale);
                    
                    float waveNoise = texture2D(noisetex, waveCoord + vec2(animateTime * 0.02, animateTime * 0.015)).x;
                    waveNoise += texture2D(noisetex, waveCoord * 2.0 - vec2(animateTime * 0.015, animateTime * 0.02)).x;
                    waveNoise = waveNoise * 0.5;
                    
                    // Modulate caustic based on wave height
                    float waveMod = 0.6 + waveNoise * 0.8;
                    caustic *= waveMod;
                    
                    float cm = clamp(distFromSurface * 0.3, 0.0, 0.8) * cs;
                    // Tint bright caustic spots teal — matches outside-water look
                    // Dark spots stay unchanged, bright spots shift toward teal/cyan
                    float tintStrength = clamp((caustic * WATER_CAUSTICS_BRIGHTNESS - 0.8) * 2.0, 0.0, 1.0);
                    vec3 causticRgb = mix(vec3(caustic * WATER_CAUSTICS_BRIGHTNESS), vec3(0.4, 1.2, 1.4) * caustic * WATER_CAUSTICS_BRIGHTNESS, tintStrength);
                    sceneColOut *= pow(mix(vec3(1.0), causticRgb, cm), vec3(WATER_CAUSTICS_POWER));
                }
            }
        }

        #if ANTI_ALIASING >= 2
            vec3 dither = fract(getRng3(screenTexelCoord & 255) + frameFract);
        #else
            vec3 dither = getRng3(screenTexelCoord & 255);
        #endif

        // Get view distance
        float viewDot = lengthSquared(viewPos);
        float viewDotInvSqrt = inversesqrt(viewDot);
        float viewDist = viewDot * viewDotInvSqrt;

        // Get normalized eyePlayerPos
        vec3 nEyePlayerPos = eyePlayerPos * viewDotInvSqrt;

        // Get fog factor
        float fogFactor = getFogFactor(viewDist, nEyePlayerPos.y, feetPlayerPos.y + cameraPosition.y);

        // Border fog
        #ifdef BORDER_FOG
            float borderFog = getBorderFog(viewDist);
        #else
            float borderFog = 0.0;
        #endif

        // Materials and programs that come after deferred mask
        vec3 matRaw0 = texelFetch(colortex3, screenTexelCoord, 0).xyz;

        // Interactive short grass
        if(matRaw0.z == 1.0){
            vec3 worldPos = feetPlayerPos + cameraPosition;
            float dist = length(worldPos.xz - cameraPosition.xz);
            if(dist < 1.0){
                // Distort the texture coordinates to simulate movement
                vec2 dir = normalize(worldPos.xz - cameraPosition.xz);
                vec2 offset = dir * (1.0 - dist) * 0.1;
                vec2 newTexCoord = texCoord + offset;
                sceneColOut = textureLod(colortex4, newTexCoord, 0).rgb;
                // Add a strong green tint and brighten
                sceneColOut *= vec3(0.5, 1.5, 0.5);
                sceneColOut += vec3(0.2);
            }
        }

        // If the object renders after deferred apply separate lighting
        if(matRaw0.z > 0 && matRaw0.z < 1){
            // Declare and get materials
            vec3 albedo = texelFetch(colortex2, screenTexelCoord, 0).rgb;
            vec3 normal = texelFetch(colortex1, screenTexelCoord, 0).xyz;

            // Apply deffered shading
            sceneColOut = complexShadingDeferred(sceneColOut, screenPos, viewPos, mat3(gbufferModelView) * normal, albedo, dither, viewDotInvSqrt, matRaw0.x, matRaw0.y, 0.0, realSky);

            // Get basic sky fog color
            vec3 fogSkyCol = getSkyFogRender(nEyePlayerPos);

            // Border fog
            #ifdef BORDER_FOG
                fogFactor = (fogFactor - 1.0) * borderFog + 1.0;
            #endif

            // Apply fog and darkness fog
            sceneColOut = ((fogSkyCol - sceneColOut) * fogFactor + sceneColOut) * getFogEffectFactor(viewDist);
        }

        // Apply darkness pulsing effect
        sceneColOut *= 1.0 - darknessLightFactor;

        #if defined WORLD_LIGHT || !defined FORCE_DISABLE_CLOUDS && CLOUD_TYPE == 2
            bool isSky = depth == 1.0;

            float feetPlayerDot = lengthSquared(feetPlayerPos);
            float feetPlayerDotInvSqrt = inversesqrt(feetPlayerDot);
            float feetPlayerDist = feetPlayerDot * feetPlayerDotInvSqrt;

            vec3 nFeetPlayerPos = feetPlayerPos * feetPlayerDotInvSqrt;
        #endif

        #ifdef WORLD_LIGHT
            // Apply volumetric light
            if(VOLUMETRIC_LIGHTING_STRENGTH != 0 && isEyeInWater != 2)
                sceneColOut += getVolumetricLight(nFeetPlayerPos, feetPlayerDist, fogFactor, borderFog, dither.x, isSky);
        #endif

        #if CLOUD_TYPE == 2 && !defined FORCE_DISABLE_CLOUDS && defined WORLD_LIGHT
            // Get the 1st layer of volumetric clouds position
            // Note that the clouds needs to move westward just as in vanilla
            vec3 cloudStartPos0 = vec3(cameraPosition.x + fragmentFrameTime, cameraPosition.y - volumetricCloudHeight, cameraPosition.z);

            // Get volumetric clouds (returns vec4: .rgb=3D lit color, .a=blended opacity)
            vec4 cloudData = volumetricClouds(nFeetPlayerPos, cloudStartPos0, viewDist, dither.x, isSky);

            #ifdef DOUBLE_LAYERED_CLOUDS
                // Get the 2nd layer of volumetric clouds position by reusing the 1st layer's position
                vec3 cloudStartPos1 = vec3(cloudStartPos0.x - 2064.0, cloudStartPos0.y - SECOND_CLOUD_HEIGHT, cloudStartPos0.z - 2064.0);

                // Variate by swizzling the 2 cloud channels
                vec4 cloudData2 = volumetricClouds(nFeetPlayerPos, cloudStartPos1, viewDist, dither.x, isSky);
                cloudData = vec4(max(cloudData.rgb, cloudData2.rgb), max(cloudData.a, cloudData2.a));
            #endif

            #ifdef DYNAMIC_CLOUDS
                float fadeTime = saturate(cos(fragmentFrameTime * FADE_SPEED) + 0.5);

                float cloudFinal = mix(mix(cloudData.a, cloudData.a, fadeTime), cloudData.a, rainStrength) * 0.75;
            #else
                float cloudFinal = cloudData.a * 0.75;
            #endif

            // Base cloud sky colour
            #ifdef FORCE_DISABLE_DAY_CYCLE
                vec3 cloudSkyCol = (toLinear(nightVision * 0.5 + AMBIENT_LIGHTING) + lightningFlash) + lightCol + skyCol;
            #else
                vec3 cloudSkyCol = (toLinear(nightVision * 0.5 + AMBIENT_LIGHTING) + lightningFlash) + mix(moonCol, sunCol, dayCycleAdjust) + skyCol;
            #endif

            
            // 2D layer = normal white/sky colour
            // 3D cumulus layer = bright CYAN tint
            // Overlap = blended
            // 2D layer: lerp toward ambient sky color
            // 2D layer: lerp toward ambient sky color
            sceneColOut = mix(sceneColOut, cloudSkyCol, cloudFinal);
            // 3D cumulus layer: add lit cloud color directly
            sceneColOut += cloudData.rgb * cloudFinal * 2.0;
            // 3D cumulus layer: add lit cloud color directly (Beer-Lambert + three-zone lighting)
            sceneColOut += cloudData.rgb * cloudFinal * 2.0;
        #endif

        // Full-screen underwater distortion when the camera is submerged.
        // This provides a subtle wavy/chromatic displacement of the final image.
        #ifndef UNDERWATER_DISTORTION_STRENGTH
            const float UNDERWATER_DISTORTION_STRENGTH = 0.0025;
        #endif
        #ifndef UNDERWATER_DISTORTION_SPEED
            const float UNDERWATER_DISTORTION_SPEED = 8.0;
        #endif
        #ifndef UNDERWATER_DISTORTION_FREQ
            const float UNDERWATER_DISTORTION_FREQ = 18.0;
        #endif
        #ifndef UNDERWATER_DISTORTION_CHROMA
            const float UNDERWATER_DISTORTION_CHROMA = 0.25;
        #endif

        if(isEyeInWater == 1){
            float t = fragmentFrameTime * UNDERWATER_DISTORTION_SPEED;

            // Depth-based factor (makes effect stronger when looking deeper)
            float depthFactor = clamp((1.0 - depth) * 2.0, 0.0, 1.0);

            // Base amplitude scaled by depth for more immersion
            float amp = UNDERWATER_DISTORTION_STRENGTH * (0.6 + 0.6 * depthFactor);

            // Sinusoidal waves across screen coords to emulate surface ripples
            vec2 wave = vec2(
                sin((texCoord.y * UNDERWATER_DISTORTION_FREQ) + t),
                cos((texCoord.x * UNDERWATER_DISTORTION_FREQ) + t * 0.9)
            );

            // Slight variation using screen-space depth to decorrelate offsets
            float localDepthSample = texture(depthtex0, texCoord).x;
            vec2 baseOffset = wave * amp * (0.8 + 0.6 * localDepthSample);

            // Chromatic dispersion offsets (R displaced most, B least)
            vec2 offR = baseOffset * (1.0 + UNDERWATER_DISTORTION_CHROMA * 0.8);
            vec2 offG = baseOffset * (1.0 + UNDERWATER_DISTORTION_CHROMA * 0.2);
            vec2 offB = baseOffset * (1.0 - UNDERWATER_DISTORTION_CHROMA * 0.5);

            vec3 sampleR = texture(colortex4, texCoord + offR).rgb;
            vec3 sampleG = texture(colortex4, texCoord + offG).rgb;
            vec3 sampleB = texture(colortex4, texCoord + offB).rgb;

            vec3 distorted = vec3(sampleR.r, sampleG.g, sampleB.b);

            // Blend intensity: stronger when deeper, tuned to be subtle by default
            float blend = clamp(0.55 + depthFactor * 0.45, 0.0, 1.0) * 0.75;

            sceneColOut = mix(sceneColOut, distorted, blend);

        }



        // Clamp scene color to prevent NaNs during post processing
        sceneColOut = max(sceneColOut, vec3(0));

        // Big moving ground cloud shadow removed per user request.
    }
#endif
